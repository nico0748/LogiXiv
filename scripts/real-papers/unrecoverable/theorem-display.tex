\section{Setting}

We minimise the average $\frac{1}{M}\sum_{m=1}^{M} F_m(x)$ of the local objectives over the workers.
The dispersion term $\frac{1}{M}\sum_{m=1}^{M}\|\nabla F_m(x) - \nabla F(x)\|^2$ controls the aggregation error.
The step size is set to $\eta = \frac{1}{L}\sqrt{\frac{\sigma^2}{T}}$ in every experiment we report.

\s{Each worker \m{$m$}{m} holds a local dataset and computes a stochastic gradient with bounded variance.}
\s{The aggregation step averages the compressed residuals across the \m{$M$}{M} workers at every round.}

\begin{assumption}[Smoothness]
\s{Each \m{$F_m$}{Fm} is \m{$L$}{L}-smooth, that is \m{$\|\nabla F_m(x) - \nabla F_m(y)\| \le L\|x-y\|$}{∥∇Fm(x) − ∇Fm(y)∥ ≤ L∥x − y∥} for all \m{$x$}{x}.}
\end{assumption}

\begin{assumption}[Bounded variance]
\s{The averaged squared deviation of the local gradients from the global gradient is bounded by a constant.}
\end{assumption}

\section{Main result}
\s{Our main result bounds the residual error of the aggregation rule under the two assumptions above.}

\begin{theorem}
\s{Under Assumptions 1 and 2, the iterates of the method satisfy the bound stated below.}
\begin{equation}
\frac{1}{T}\sum_{t=0}^{T-1}\mathbb{E}\|\nabla F(x_t)\|^2 \le \frac{c\,(F(x_0)-F^\star)}{\eta T} + c\,\eta L \sigma^2 ,
\end{equation}
\s{Here \m{$c > 0$}{c > 0} is an absolute constant that does not depend on \m{$T$}{T}.}
\end{theorem}

\begin{lemma}
\s{For all \m{$t$}{t} larger than \m{$t_0 = \lceil 2/\delta \rceil$}{t0 = ⌈2/δ⌉} we have the lower bound stated below.}
\begin{equation}
\mathbb{E}\|\bar{e}_t\|^2 \ge \frac{\eta^2 (1-\delta)}{4}\,\zeta^2 .
\end{equation}
\end{lemma}

\s{The proof is deferred to Appendix A, and the constants are not optimised.}

\section{Discussion}
\s{The bound degrades gracefully as the compression ratio grows towards one half of the coordinates.}
\s{Collecting the surviving terms gives the claimed lower bound on the expected residual norm.}
\s{We expect the same argument to apply to the momentum variant without further modification.}
